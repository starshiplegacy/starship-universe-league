/* ============================================================
 UTILIDADES
 ============================================================ */
const stage           = document.getElementById('stage');
const submenuEl = document.getElementById('submenu');
const resultEl = document.getElementById('result');
const statusEl = document.querySelector('.status');


let abierto = null;


function mostrarResultado(texto) {
resultEl.textContent = texto;
resultEl.classList.add('show');
clearTimeout(mostrarResultado._t);
mostrarResultado._t = setTimeout(() => resultEl.classList.remove('show'), 2000);

}


function actualizarEstado() {
const rel = relacion.conocidos
    ? (relacion.romance > 0 ? 'Interés romántico' : 'Conocidos')
    : 'Desconocidos';
statusEl.innerHTML =
    'Relación: <span>' + rel + '</span><br>' +
    'Amistad: <span>' + relacion.amistad + '</span> · ' +
    'Romance: <span>' + relacion.romance + '</span>';
}


// ¿Está disponible este ítem según la relación?
function disponible(item) {
if (!item.requiere) return true;
if (item.requiere === 'conocidos') return relacion.conocidos;
if (item.requiere === 'amistad') return relacion.amistad >= 20;
if (item.requiere === 'romance') return relacion.romance >= 20;
return true;
}


/* ============================================================
    ABRIR / CERRAR SUBMENÚ
    ============================================================ */
function cerrarSubmenu() {
submenuEl.classList.remove('show');
if (abierto) abierto.classList.remove('open');
abierto = null;
}


function abrirSubmenu(bubble) {

const key = bubble.dataset.menu;
const menu = menus[key];
if (!menu) return;


// si ya estaba abierto este, cerrar
if (abierto === bubble) { cerrarSubmenu(); return; }


cerrarSubmenu();
abierto = bubble;
bubble.classList.add('open');


submenuEl.innerHTML = '';


const head = document.createElement('div');
head.className = 'hint';
head.textContent = menu.titulo;
submenuEl.appendChild(head);


menu.items.forEach(item => {
 const btn = document.createElement('button');
 btn.className = 'item' + (disponible(item) ? '' : ' locked');
 btn.innerHTML =
     '<span>' + item.icon + '</span><span>' + item.label + '</span>' +
     (disponible(item) ? '' : '<span class="lock">▸</span>');


 if (disponible(item)) {
     btn.addEventListener('click', e => {
      e.stopPropagation();
      aplicarEfecto(item);
     });
 }

    submenuEl.appendChild(btn);
});


// posicionar el submenú junto a la burbuja
const b = bubble.getBoundingClientRect();
const s = stage.getBoundingClientRect();


let left = b.right - s.left + 10;
let top = b.top - s.top;


submenuEl.classList.add('show');


// evitar que se salga del marco
const w = submenuEl.offsetWidth;
const h = submenuEl.offsetHeight;
if (left + w > s.width - 8) left = b.left - s.left - w - 10;
if (left < 8) left = 8;
if (top + h > s.height - 8) top = s.height - h - 8;
if (top < 8) top = 8;


submenuEl.style.left = left + 'px';
submenuEl.style.top = top + 'px';
}


/* ============================================================
    APLICAR EFECTO
    ============================================================ */
function aplicarEfecto(item) {
const e = item.efecto || {};


if (e.amistad) relacion.amistad += e.amistad;

if (e.romance) relacion.romance += e.romance;
if (e.conocidos) relacion.conocidos = true;


// límites
relacion.amistad = Math.max(-100, Math.min(100, relacion.amistad));
relacion.romance = Math.max(0, Math.min(100, relacion.romance));


actualizarEstado();


// mensaje
let msg = item.label;
if (e.conocidos && !relacion.conocidos) msg += ' — ahora se conocen';
if (e.amistad) msg += ' (' + (e.amistad > 0 ? '+' : '') + e.amistad + ' amistad)';
if (e.romance) msg += ' (' + (e.romance > 0 ? '+' : '') + e.romance + ' romance)';


mostrarResultado(msg);
cerrarSubmenu();
}


/* ============================================================
    EVENTOS
    ============================================================ */
document.querySelectorAll('.bubble').forEach(b => {
b.addEventListener('click', e => {
    e.stopPropagation();
    abrirSubmenu(b);
});
});


// clic fuera → cerrar
stage.addEventListener('click', e => {

if (e.target.closest('.submenu') || e.target.closest('.bubble')) return;
cerrarSubmenu();
});


document.addEventListener('keydown', e => {
if (e.key === 'Escape') cerrarSubmenu();
});


actualizarEstado();
</script>
