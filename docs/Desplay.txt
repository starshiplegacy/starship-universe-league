<script>
/* ============================================================
 ESTADO
 ============================================================ */
const lyries = {
lyrie1: { name: 'Lyrie', el: document.getElementById('lyrie1') },
lyrie2: { name: 'Nayeon', el: document.getElementById('lyrie2') }
};
let selected = 'lyrie1';
let destino = null; // { id, nombre, x, y }


const phone        = document.getElementById('phone');
const map        = document.getElementById('map');
const sheet      = document.getElementById('sheet');
const sheetHead = document.getElementById('sheetHead');
const sheetBody = document.getElementById('sheetBody');
const toastEl = document.getElementById('toast');

/* ============================================================
    UTILIDADES
    ============================================================ */
function showToast(text) {
toastEl.textContent = text;
toastEl.classList.add('show');
clearTimeout(showToast._t);
showToast._t = setTimeout(() => toastEl.classList.remove('show'), 1800);
}


function closeSheet() {
sheet.classList.remove('open');
destino = null;
}


// Centro de un elemento relativo al mapa
function centerOf(el) {
const m = map.getBoundingClientRect();
const r = el.getBoundingClientRect();
return {
    x: r.left - m.left + r.width / 2,
    y: r.top - m.top + r.height / 2
};
}


// Mueve un Lyrie a un punto (con pequeño desfase si van dos)
function walkTo(id, x, y, offsetX = 0, offsetY = 0) {
const l = lyries[id];
l.el.style.left = (x + offsetX - 17) + 'px';
l.el.style.top = (y + offsetY - 17) + 'px';
}

// Lista de acompañantes posibles
function companions() {
return Object.keys(lyries)
    .filter(id => id !== selected)
    .map(id => ({ id, name: lyries[id].name }));
}


/* ============================================================
    HOJA: ELEGIR SOLO O CON ALGUIEN
    ============================================================ */
function openTravelSheet(placeName) {
sheetHead.textContent = 'Ir a ' + placeName;
sheetBody.innerHTML = '';


// --- opción 1: solo ---
const solo = document.createElement('button');
solo.className = 'opt solo';
solo.innerHTML = '<span class="ico">�</span> Irse solo';
solo.addEventListener('click', () => {
    const c = centerOf(destino.el);
    walkTo(selected, c.x, c.y);
    closeSheet();
    showToast(lyries[selected].name + ' va a ' + placeName);
});
sheetBody.appendChild(solo);


// --- opción 2: con alguien ---
const withLabel = document.createElement('div');
withLabel.className = 'sub';
withLabel.textContent = 'Ir con alguien';

sheetBody.appendChild(withLabel);


const lista = companions();
if (lista.length === 0) {
    const vacio = document.createElement('div');
    vacio.className = 'empty';
    vacio.textContent = 'No hay otros Lyries cerca para acompañarte.';
    sheetBody.appendChild(vacio);
} else {
    lista.forEach(c => {
     const btn = document.createElement('button');
     btn.className = 'opt';
     btn.innerHTML = '<span class="ico">�</span> Ir con ' + c.name;
     btn.addEventListener('click', () => {
      const p = centerOf(destino.el);
      walkTo(selected, p.x, p.y, -14, 0);
      walkTo(c.id,    p.x, p.y, 14, 0);
      closeSheet();
      showToast(lyries[selected].name + ' y ' + c.name + ' van a ' + placeName);
     });
     sheetBody.appendChild(btn);
    });
}


sheet.classList.add('open');
}


/* ============================================================
    CLICK EN UN LUGAR DEL MAPA
    ============================================================ */
['casa', 'parque', 'tienda', 'piscina', 'playa'].forEach(id => {

const el = document.getElementById(id);
el.addEventListener('click', e => {
  e.stopPropagation();
  destino = { id, el, nombre: el.textContent.trim() };
  openTravelSheet(destino.nombre);
});
});


/* ============================================================
 CLICK EN EL MAPA VACÍO → también desplazarse
 ============================================================ */
map.addEventListener('click', e => {
if (e.target.closest('.place') || e.target.closest('.lyrie')) return;


const m = map.getBoundingClientRect();
const x = e.clientX - m.left;
const y = e.clientY - m.top;


destino = {
  id: 'punto',
  el: null,
  nombre: 'ese punto',
  x, y
};


sheetHead.textContent = 'Ir a ese punto';
sheetBody.innerHTML = '';


const solo = document.createElement('button');
solo.className = 'opt solo';
solo.innerHTML = '<span class="ico">�</span> Irse solo';

solo.addEventListener('click', () => {
 walkTo(selected, x, y);
 closeSheet();
 showToast(lyries[selected].name + ' se desplaza');
});
sheetBody.appendChild(solo);


const withLabel = document.createElement('div');
withLabel.className = 'sub';
withLabel.textContent = 'Ir con alguien';
sheetBody.appendChild(withLabel);


const lista = companions();
if (lista.length === 0) {
 const vacio = document.createElement('div');
 vacio.className = 'empty';
 vacio.textContent = 'No hay otros Lyries cerca para acompañarte.';
 sheetBody.appendChild(vacio);
} else {
 lista.forEach(c => {
  const btn = document.createElement('button');
  btn.className = 'opt';
  btn.innerHTML = '<span class="ico">�</span> Ir con ' + c.name;
  btn.addEventListener('click', () => {
      walkTo(selected, x, y, -14, 0);
      walkTo(c.id,    x, y, 14, 0);
      closeSheet();
      showToast(lyries[selected].name + ' y ' + c.name + ' se desplazan');
  });
  sheetBody.appendChild(btn);
 });

}


sheet.classList.add('open');
});


/* ============================================================
 SELECCIÓN DE LYRIE
 ============================================================ */
Object.keys(lyries).forEach(id => {
lyries[id].el.addEventListener('click', e => {
    e.stopPropagation();
    closeSheet();
    document.querySelectorAll('.lyrie').forEach(el => el.style.outline = '');
    lyries[id].el.style.outline = '3px solid #ffd400';
    lyries[id].el.style.outlineOffset = '2px';
    selected = id;
    showToast('Controlas a ' + lyries[id].name);
});
});


/* ============================================================
 CERRAR
 ============================================================ */
document.getElementById('closeAll').addEventListener('click', () => {
closeSheet();
showToast('Desplazamiento cerrado');
});


document.addEventListener('keydown', e => {
if (e.key === 'Escape') closeSheet();
});

lyries.lyrie1.el.style.top = '300px';
lyries.lyrie2.el.style.left = '190px';
lyries.lyrie2.el.style.top = '330px';
</script>
