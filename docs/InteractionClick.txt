<script>
/* ============================================================
ESTADO DEL JUEGO
============================================================ */

const state = {
selected: 'lyrie1',         // Lyrie controlado
lyries: {
     lyrie1: { name: 'Lyrie', el: null },
     lyrie2: { name: 'Nayeon', el: null }
}
};
state.lyries.lyrie1.el = document.getElementById('lyrie1');
state.lyries.lyrie2.el = document.getElementById('lyrie2');


const game         = document.getElementById('game');
const ctx        = document.getElementById('ctx');
const ctxTitle = document.getElementById('ctxTitle');
const ctxItems = document.getElementById('ctxItems');
const submenu = document.getElementById('submenu');
const subTitle = document.getElementById('subTitle');
const subItems = document.getElementById('subItems');
const toastEl = document.getElementById('toast');


/* ============================================================
    UTILIDADES
    ============================================================ */
function showToast(text) {
toastEl.textContent = text;
toastEl.classList.add('show');
clearTimeout(showToast._t);
showToast._t = setTimeout(() => toastEl.classList.remove('show'), 1600);
}


function closeMenus() {
ctx.classList.remove('open');

submenu.classList.remove('open');
}


// Abre el menú contextual en coordenadas del contenedor
function openMenu(title, items, x, y) {
ctxTitle.textContent = title;
ctxItems.innerHTML = '';


items.forEach(item => {
    const btn = document.createElement('button');
    btn.textContent = item.label;
    btn.addEventListener('click', e => {
     e.stopPropagation();
     if (item.submenu) {
         openSubmenu(item.submenuTitle || 'Elegir', item.submenu, x, y);
     } else {
         closeMenus();
         item.action && item.action();
     }
    });
    ctxItems.appendChild(btn);
});


// Ajuste para que no se salga del contenedor
const rect = game.getBoundingClientRect();
let left = x + 10;
let top = y + 10;
if (left + 200 > rect.width) left = x - 200;
if (top + 260 > rect.height) top = Math.max(6, rect.height - 270);


ctx.style.left = left + 'px';

ctx.style.top = top + 'px';
ctx.classList.add('open');
submenu.classList.remove('open');
}


function openSubmenu(title, options, x, y) {
subTitle.textContent = title;
subItems.innerHTML = '';


options.forEach(opt => {
    const btn = document.createElement('button');
    btn.textContent = opt.label;
    btn.addEventListener('click', e => {
     e.stopPropagation();
     closeMenus();
     opt.action && opt.action();
    });
    subItems.appendChild(btn);
});


const rect = game.getBoundingClientRect();
let left = x + 210;
let top = y + 10;
if (left + 180 > rect.width) left = Math.max(6, x - 190);
if (top + 200 > rect.height) top = Math.max(6, rect.height - 210);


submenu.style.left = left + 'px';
submenu.style.top = top + 'px';
submenu.classList.add('open');
}

// Mueve un Lyrie hacia un elemento (centro del objeto)
function moveTo(lyrieId, targetEl, label) {
const l = state.lyries[lyrieId];
const g = game.getBoundingClientRect();
const t = targetEl.getBoundingClientRect();


const x = t.left - g.left + t.width / 2 - 23;
const y = t.top - g.top + t.height / 2 - 23;


l.el.style.left = x + 'px';
l.el.style.top = y + 'px';


if (label) showToast(l.name + ': ' + label);
}


// Mueve a un punto del suelo
function moveToPoint(lyrieId, x, y, label) {
const l = state.lyries[lyrieId];
l.el.style.left = (x - 23) + 'px';
l.el.style.top = (y - 23) + 'px';
if (label) showToast(l.name + ': ' + label);
}


// Lista de otros Lyries (para "con quién")
function otherLyries() {
return Object.keys(state.lyries)
    .filter(id => id !== state.selected)
    .map(id => ({ label: state.lyries[id].name, id }));
}


/* ============================================================

 SELECCIÓN DE PERSONAJE
 ============================================================ */
Object.keys(state.lyries).forEach(id => {
state.lyries[id].el.addEventListener('click', e => {
  e.stopPropagation();
  closeMenus();
  document.querySelectorAll('.lyrie').forEach(el => el.classList.remove('selected'));
  state.lyries[id].el.classList.add('selected');
  state.selected = id;
});
});


/* ============================================================
 CLICK EN OBJETOS
 ============================================================ */
// VASO → Limpiar / Guardar
document.getElementById('vaso').addEventListener('click', e => {
e.stopPropagation();
const x = e.offsetX, y = e.offsetY;
openMenu('Vaso', [
  {
      label: 'Limpiar',
      action: () => moveTo(state.selected, document.getElementById('fregadero'), 'lavando el
vaso')
  },
  {
      label: 'Guardar',
      action: () => moveTo(state.selected, document.getElementById('nevera'), 'guardando el
vaso')
  }
], x + 40, y + 60);

});


// NEVERA y FREGADERO: acciones simples
document.getElementById('nevera').addEventListener('click', e => {
e.stopPropagation();
const x = e.offsetX, y = e.offsetY;
openMenu('Nevera', [
     { label: 'Abrir', action: () => moveTo(state.selected, e.currentTarget, 'abriendo la nevera')
},
     { label: 'Guardar', action: () => moveTo(state.selected, e.currentTarget, 'guardando algo') }
], x + 40, y + 60);
});


document.getElementById('fregadero').addEventListener('click', e => {
e.stopPropagation();
const x = e.offsetX, y = e.offsetY;
openMenu('Fregadero', [
     { label: 'Limpiar', action: () => moveTo(state.selected, e.currentTarget, 'limpiando') },
     { label: 'Lavarse las manos', action: () => moveTo(state.selected, e.currentTarget,
'lavándose las manos') }
], x + 40, y + 60);
});


// PISCINA y MAR → Ir aquí / Ir aquí con...
['piscina', 'mar'].forEach(id => {
document.getElementById(id).addEventListener('click', e => {
     e.stopPropagation();
     const el = e.currentTarget;
     const x = e.offsetX, y = e.offsetY;
     const nombre = id === 'piscina' ? 'Piscina' : 'Mar';
     const verb = id === 'piscina' ? 'nadando' : 'en el mar';

  openMenu(nombre, [
      {
          label: 'Ir aquí',
          action: () => moveTo(state.selected, el, verb)
      },
      {
          label: 'Ir aquí con...',
          submenuTitle: '¿Con quién?',
          submenu: otherLyries().map(o => ({
           label: o.label,
           action: () => {
               moveTo(state.selected, el, verb);
               setTimeout(() => moveTo(o.id, el, verb), 120);
           }
          }))
      }
  ], x + 100, y + 60);
});
});


/* ============================================================
 CLICK EN OTRO LYRIE (misma unidad)
 ============================================================ */
Object.keys(state.lyries).forEach(id => {
state.lyries[id].el.addEventListener('click', e => {
  e.stopPropagation();
  if (id === state.selected) return; // ya está seleccionado


  const otro = state.lyries[id];
  const x = e.offsetX, y = e.offsetY;

const hablarDe = (bien) => ({
 label: bien ? 'Hablar bien de...' : 'Hablar mal de...',
 submenuTitle: bien ? '¿De quién bien?' : '¿De quién mal?',
 submenu: otherLyries()
  .filter(o => o.id !== id)
  .map(o => ({
      label: o.label,
      action: () => showToast(
          state.lyries[state.selected].name + ' habla ' + (bien ? 'bien' : 'mal') +
          ' de ' + o.label
      )
  }))
});


openMenu(otro.name, [
 { label: 'Controlar este Lyrie', action: () => {
      document.querySelectorAll('.lyrie').forEach(el => el.classList.remove('selected'));
      otro.el.classList.add('selected');
      state.selected = id;
      showToast('Ahora controlas a ' + otro.name);
  }
 },
 { label: 'Hablar con ' + otro.name, action: () => {
      moveTo(state.selected, otro.el, 'hablando con ' + otro.name);
  }
 },
 { label: 'Pelear con ' + otro.name, action: () => {
      moveTo(state.selected, otro.el, 'peleando con ' + otro.name);
  }
 },

      hablarDe(true),
      hablarDe(false),
      { label: 'Cotillear', action: () => {
              moveTo(state.selected, otro.el, 'cotilleando');
          }
      }
  ], x + 40, y + 40);
});
});


/* ============================================================
 CLICK EN EL SUELO
 ============================================================ */
game.addEventListener('click', e => {
// Si el click fue sobre un menú, no hacer nada
if (e.target.closest('.ctx') || e.target.closest('.submenu')) return;


closeMenus();


const rect = game.getBoundingClientRect();
const x = e.clientX - rect.left;
const y = e.clientY - rect.top;


openMenu('Suelo', [
  {
      label: 'Ir aquí',
      action: () => moveToPoint(state.selected, x, y, 'caminando')
  },
  {
      label: 'Ir aquí con...',
      submenuTitle: '¿Con quién?',

      submenu: otherLyries().map(o => ({
       label: o.label,
       action: () => {
           moveToPoint(state.selected, x, y, 'caminando');
           setTimeout(() => moveToPoint(o.id, x + 50, y + 20, 'caminando'), 120);
       }
      }))
  }
], x, y);
});


// Cerrar menús al hacer click fuera (en el body)
document.addEventListener('click', e => {
if (!e.target.closest('.game')) closeMenus();
});


// ESC cierra los menús
document.addEventListener('keydown', e => {
if (e.key === 'Escape') closeMenus();
});
</script>
